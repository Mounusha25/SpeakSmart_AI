class HomePage extends StatefulWidget {
  const HomePage({super.key});

  @override
  State<HomePage> createState() => _HomePageState();
}

class _HomePageState extends State<HomePage>
    with SingleTickerProviderStateMixin {
  final _rng = Random();
  final List<_HistoryItem> _history = [];

  // Local pools
  final List<String> _jokesPool = [];
  final List<String> _advicePool = [];

  late final GeminiService _gemini;
  late TabController _tabController;

  bool _loadingJokes = false;
  bool _loadingAdvice = false;
  bool _loadingStory = false;
  bool _loadingMeme = false;

  // Controllers for story and meme
  final TextEditingController _storyKeywordsController =
      TextEditingController();
  final TextEditingController _memeTextController = TextEditingController();

  // Firestore references
  late final CollectionReference<Map<String, dynamic>> _jokesRef;
  late final CollectionReference<Map<String, dynamic>> _adviceRef;
  late final CollectionReference<Map<String, dynamic>> _storiesRef;
  late final CollectionReference<Map<String, dynamic>> _memesRef;

  @override
  void initState() {
    super.initState();
    _gemini = GeminiService();
    _tabController = TabController(length: 2, vsync: this);

    final db = FirebaseFirestore.instance;
    _jokesRef = db.collection('jokes');
    _adviceRef = db.collection('advice');
    _storiesRef = db.collection('stories');
    _memesRef = db.collection('memes');
  }

  @override
  void dispose() {
    _tabController.dispose();
    _storyKeywordsController.dispose();
    _memeTextController.dispose();
    super.dispose();
  }

  Future<void> _addRandom(bool isJoke) async {
    setState(() {
      if (isJoke) {
        _loadingJokes = _jokesPool.isEmpty;
      } else {
        _loadingAdvice = _advicePool.isEmpty;
      }
    });

    try {
      if (isJoke && _jokesPool.isEmpty) {
        final batch = await _gemini.fetchBatch(kind: PromptKind.joke);
        _jokesPool.addAll(batch);
      } else if (!isJoke && _advicePool.isEmpty) {
        final batch = await _gemini.fetchBatch(kind: PromptKind.advice);
        _advicePool.addAll(batch);
      }

      final pool = isJoke ? _jokesPool : _advicePool;
      if (pool.isEmpty) {
        _toast("No items returned. Try again.");
        return;
      }

      final text = pool[_rng.nextInt(pool.length)];
      final now = DateTime.now();

      setState(() {
        _history.insert(
          0,
          _HistoryItem(
            text: text,
            type: isJoke ? ItemType.joke : ItemType.advice,
            time: now,
          ),
        );
      });

      await _writeToFirestore(
        collection: isJoke ? _jokesRef : _adviceRef,
        text: text,
        createdAt: now,
      );
    } catch (e) {
      _toast("Error: $e");
    } finally {
      if (mounted) {
        setState(() {
          if (isJoke) _loadingJokes = false;
          if (!isJoke) _loadingAdvice = false;
        });
      }
    }
  }

  Future<void> _generateStory() async {
    final keywords = _storyKeywordsController.text.trim();
    if (keywords.isEmpty) {
      _toast("Please enter keywords for the story");
      return;
    }

    setState(() => _loadingStory = true);

    try {
      final story = await _gemini.generateStory(keywords);
      final now = DateTime.now();

      setState(() {
        _history.insert(
          0,
          _HistoryItem(
            text: story,
            type: ItemType.story,
            time: now,
            metadata: keywords,
          ),
        );
      });

      await _writeToFirestore(
        collection: _storiesRef,
        text: story,
        createdAt: now,
        metadata: {'keywords': keywords},
      );

      _storyKeywordsController.clear();
      _toast("Story generated successfully!");
    } catch (e) {
      _toast("Error generating story: $e");
    } finally {
      if (mounted) setState(() => _loadingStory = false);
    }
  }

  Future<void> _generateMeme() async {
    final text = _memeTextController.text.trim();
    if (text.isEmpty) {
      _toast("Please enter text for the meme");
      return;
    }

    setState(() => _loadingMeme = true);

    try {
      final memeUrl = await _gemini.generateMemeImage(text);
      final now = DateTime.now();

      setState(() {
        _history.insert(
          0,
          _HistoryItem(
            text: text,
            type: ItemType.meme,
            time: now,
            imageUrl: memeUrl,
          ),
        );
      });

      await _writeToFirestore(
        collection: _memesRef,
        text: text,
        createdAt: now,
        metadata: {'imageUrl': memeUrl},
      );

      _memeTextController.clear();
      _toast("Meme generated successfully!");
    } catch (e) {
      _toast("Error generating meme: $e");
    } finally {
      if (mounted) setState(() => _loadingMeme = false);
    }
  }

  Future<void> _writeToFirestore({
    required CollectionReference<Map<String, dynamic>> collection,
    required String text,
    required DateTime createdAt,
    Map<String, dynamic>? metadata,
  }) async {
    await collection.add({
      'text': text,
      'ts': FieldValue.serverTimestamp(),
      'clientTime': createdAt.toIso8601String(),
      'source': 'gemini-2.0-flash',
      'length': text.length,
      if (metadata != null) ...metadata,
    });
  }

  void _toast(String msg) {
    ScaffoldMessenger.of(context).showSnackBar(
      SnackBar(
        content: Text(msg),
        behavior: SnackBarBehavior.floating,
        shape: RoundedRectangleBorder(borderRadius: BorderRadius.circular(10)),
      ),
    );
  }

  @override
  Widget build(BuildContext context) {
    final busy =
        _loadingJokes || _loadingAdvice || _loadingStory || _loadingMeme;
    final theme = Theme.of(context);

    return Scaffold(
      appBar: AppBar(
        title: const Text(
          'Creative Companion',
          style: TextStyle(fontWeight: FontWeight.bold),
        ),
        centerTitle: true,
        elevation: 0,
        actions: [
          IconButton(
            tooltip: 'Refill pools',
            onPressed: busy
                ? null
                : () async {
                    setState(() {
                      _loadingJokes = true;
                      _loadingAdvice = true;
                    });
                    try {
                      final jokes = await _gemini.fetchBatch(
                        kind: PromptKind.joke,
                      );
                      final tips = await _gemini.fetchBatch(
                        kind: PromptKind.advice,
                      );
                      setState(() {
                        _jokesPool
                          ..clear()
                          ..addAll(jokes);
                        _advicePool
                          ..clear()
                          ..addAll(tips);
                      });
                      _toast("Pools refilled!");
                    } catch (e) {
                      _toast("Refill failed: $e");
                    } finally {
                      if (mounted) {
                        setState(() {
                          _loadingJokes = false;
                          _loadingAdvice = false;
                        });
                      }
                    }
                  },
            icon: const Icon(Icons.refresh_rounded),
          ),
        ],
        bottom: TabBar(
          controller: _tabController,
          tabs: const [
            Tab(icon: Icon(Icons.dashboard_rounded), text: 'Quick Actions'),
            Tab(icon: Icon(Icons.auto_awesome), text: 'Create'),
          ],
        ),
      ),
      body: TabBarView(
        controller: _tabController,
        children: [_buildQuickActionsTab(theme), _buildCreateTab(theme)],
      ),
    );
  }

class _ActionCard extends StatelessWidget {

  final String label;

  final VoidCallback onTap;

  const _ActionCard({
    required this.icon,
    required this.label,
    required this.color,
    required this.isLoading,
    required this.onTap,

    return Card(
      elevation: 4,

        borderRadius: BorderRadius.circular(16),
        child: Padding(
          padding: const EdgeInsets.all(20),
          child: Column(
            children: [
              Container(
                padding: const EdgeInsets.all(16),

                  shape: BoxShape.circle,

                child: isLoading

                        width: 32,

              Text(
                label,

                style: const TextStyle(

                  fontSize: 13,
                ),
              ),
            ],
          ),
        ),
      ),
    );
  }
}

class _HistoryCard extends StatelessWidget {
  final _HistoryItem item;

  const _HistoryCard({required this.item});

  @override
  Widget build(BuildContext context) {
    final theme = Theme.of(context);

    return Card(
      margin: const EdgeInsets.only(bottom: 12),
      child: Column(
        crossAxisAlignment: CrossAxisAlignment.start,
        children: [
          ListTile(
            leading: CircleAvatar(
              backgroundColor: _getColor(item.type).withOpacity(0.2),
              child: Icon(_getIcon(item.type), color: _getColor(item.type)),
            ),
            title: Text(
              _getTypeLabel(item.type),
              style: const TextStyle(fontWeight: FontWeight.bold),
            ),
            subtitle: Text(_formatTime(item.time)),
          ),
          Padding(
            padding: const EdgeInsets.fromLTRB(16, 0, 16, 16),
            child: Column(
              crossAxisAlignment: CrossAxisAlignment.start,
              children: [
                if (item.metadata != null && item.type == ItemType.story)
                  Padding(
                    padding: const EdgeInsets.only(bottom: 8),
                    child: Chip(
                      label: Text('Keywords: ${item.metadata}'),
                      backgroundColor: Colors.purple.withOpacity(0.1),
                      labelStyle: const TextStyle(fontSize: 12),
                    ),
                  ),
                Text(item.text, style: theme.textTheme.bodyMedium),
                if (item.imageUrl != null) ...[
                  const SizedBox(height: 12),
                  ClipRRect(
                    borderRadius: BorderRadius.circular(12),
                    child: Image.network(
                      item.imageUrl!,
                      fit: BoxFit.cover,
                      loadingBuilder: (context, child, loadingProgress) {
                        if (loadingProgress == null) return child;
                        return const Center(
                          child: Padding(
                            padding: EdgeInsets.all(20),
                            child: CircularProgressIndicator(),
                          ),
                        );
                      },
                      errorBuilder: (context, error, stackTrace) {
                        return Container(
                          padding: const EdgeInsets.all(20),
                          color: Colors.grey[200],
                          child: const Center(
                            child: Text('Failed to load image'),
                          ),
                        );
                      },
                    ),
                  ),
                ],
              ],
            ),
          ),
        ],
      ),
    );
  }

  IconData _getIcon(ItemType type) {
    switch (type) {
      case ItemType.joke:
        return Icons.emoji_emotions_rounded;
      case ItemType.advice:
        return Icons.lightbulb_rounded;
      case ItemType.story:
        return Icons.auto_stories_rounded;
      case ItemType.meme:
        return Icons.image_rounded;
    }
  }

  Color _getColor(ItemType type) {
    switch (type) {
      case ItemType.joke:
        return Colors.amber;
      case ItemType.advice:
        return Colors.blue;
      case ItemType.story:
        return Colors.purple;
      case ItemType.meme:
        return Colors.orange;
    }
  }

  String _getTypeLabel(ItemType type) {
    switch (type) {
      case ItemType.joke:
        return 'Joke';
      case ItemType.advice:
        return 'Advice';
      case ItemType.story:
        return 'Story';
      case ItemType.meme:
        return 'Meme';
    }
  }

  String _formatTime(DateTime t) {
    final hh = t.hour.toString().padLeft(2, '0');
    final mm = t.minute.toString().padLeft(2, '0');
    return '$hh:$mm';
  }
}

class GeminiService {
  GeminiService()
    : _model = GenerativeModel(
        model: 'gemini-2.0-flash',
        apiKey: "AIzaSyDe05JigE1kDO-Ue40jKwZtujysYE_Wlwc",
        generationConfig: GenerationConfig(temperature: 1.0),
      );

  final GenerativeModel _model;

  Future<List<String>> fetchBatch({required PromptKind kind}) async {
    final systemGoal = kind == PromptKind.joke
        ? "You are a witty, wholesome comedian. Avoid offensive or unsafe content."
        : "You are a practical, concise coach. Avoid medical/legal/financial specifics.";

    final userAsk = kind == PromptKind.joke
        ? "Give me 10 short, family-friendly one-liner jokes."
        : "Give me 10 short, practical life advice lines.";

    final formatRule =
        "Return ONLY a valid JSON array of 10 strings. No code fences, no commentary.";

    final prompt = "$systemGoal\n$userAsk\n$formatRule";

    final resp = await _model.generateContent([Content.text(prompt)]);
    final text = resp.text?.trim();
    if (text == null || text.isEmpty) {
      throw Exception("Empty response from model.");
    }

    final cleaned = _stripFences(text);
    final decoded = jsonDecode(cleaned);
    if (decoded is! List) {
      throw Exception("Unexpected format (not a JSON array).");
    }

    final items = decoded
        .where((e) => e is String && (e).trim().isNotEmpty)
        .map((e) => (e as String).trim())
        .toList();

    if (items.isEmpty) {
      throw Exception("No items parsed from response.");
    }
    return items;
  }

  Future<String> generateStory(String keywords) async {
    final prompt =
        """You are a creative storyteller. Generate a compelling, engaging story based on the following keywords: $keywords
    
The story should be:
- Between 200-400 words
- Have a clear beginning, middle, and end
- Be family-friendly and engaging
- Creatively incorporate all the keywords

Return ONLY the story text, no additional commentary.""";

    final resp = await _model.generateContent([Content.text(prompt)]);
    final text = resp.text?.trim();
    if (text == null || text.isEmpty) {
      throw Exception("Empty response from model.");
    }

    return text;
  }

  Future<String> generateMemeImage(String text) async {
    // Using Pollinations AI API for free meme/image generation
    final encodedText = Uri.encodeComponent(text);
    final seed = DateTime.now().millisecondsSinceEpoch;

    // Generate a meme-style image with text overlay
    final url =
        'https://image.pollinations.ai/prompt/$encodedText'
        '?width=800'
        '&height=600'
        '&enhance=true'
        '&nologo=true'
        '&seed=$seed';

    // Verify the image is accessible
    final response = await http
        .get(Uri.parse(url))
        .timeout(const Duration(seconds: 30));

    if (response.statusCode == 200) {
      return url;
    } else {
      throw Exception("Failed to generate meme image");
    }
  }

  String _stripFences(String s) {
    final fence = RegExp(r'^``````$');
    final m = fence.firstMatch(s);
    return m != null ? m.group(1)! : s;
  }
}
